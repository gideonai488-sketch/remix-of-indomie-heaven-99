Realtime
supabase.channel(`order:${id}`)
  .on('postgres_changes', { event: 'UPDATE', schema: 'public', table: 'orders', filter: `id=eq.${id}` }, p => setOrder(p.new))
  .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'driver_locations', filter: `rider_id=eq.${riderId}` }, p => setLoc(p.new))
  .subscribe()
